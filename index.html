HTML — markup 
<div class="hero-media">
  <!-- แบบรูปภาพ (responsive ต่างไฟล์ต่อ breakpoint) -->
  <picture class="hero-picture">
    <source
      media="(max-width:767px)"
      srcset="/images/about-mobile.gif">
    <source
      media="(max-width:1023px)"
      srcset="/images/about-tablet.gif">
    <img
      src="/images/about-desktop.gif"
      alt="CTW Cargo">
  </picture>

  <!-- หรือแบบวิดีโอ (ไฟล์เดียว ครอปด้วย CSS) -->
  <!--
  <video class="hero-video" autoplay muted loop playsinline>
    <source src="/videos/about-desktop.mp4" type="video/mp4">
  </video>
  -->
</div>
CSS — aspect-ratio ตาม breakpoint 
.hero-media{
    width:100%;
    border-radius:20px;
    overflow:hidden;
    position:relative;
}

/* Desktop */
.hero-picture,
.hero-video{
    display:block;
    width:100%;
    aspect-ratio:16/6;
}
.hero-picture img,
.hero-video{
    width:100%;
    height:100%;
    object-fit:cover;
    display:block;
}

/* Tablet */
@media (max-width:1023px){
  .hero-picture,
  .hero-video{
      aspect-ratio:16/8;
  }
}

/* Mobile */
@media (max-width:767px){
  .hero-picture,
  .hero-video{
      aspect-ratio:4/3;
  }
}
